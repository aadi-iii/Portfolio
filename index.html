// Love Calculator ❤️ - Gamified Multi-Challenge Scoring Engine
function calculateEarnedLoveScore(gameScores) {
  const { memoryPairsScore, wordScrambleTime, cupidsArrowAccuracy } = gameScores;
  const totalEarned = (memoryPairsScore * 0.35) + 
                      (wordScrambleTime * 0.30) + 
                      (cupidsArrowAccuracy * 0.35);
  return Math.min(Math.round(totalEarned), 100);
}
